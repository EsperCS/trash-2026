\section{Literature Review}

\begin{frame}{Literature Review 1: Narrow AI Calibration}
\textbf{Vấn đề chính:} neural networks hiện đại thường đạt accuracy cao nhưng confidence không phản ánh đúng xác suất đúng.

\vspace{0.2cm}
\textbf{Nhóm phương pháp đã review:}
\begin{itemize}
\item \textbf{Post-hoc calibration:} Temperature Scaling, Isotonic Regression, Beta Calibration, Bayesian Binning.
\item \textbf{Training-time calibration:} Label Smoothing, Confidence Penalty, MMCE, Focal Loss, Mixup, Deep Ensembles, MC Dropout.
\item \textbf{Metrics:} ECE, Adaptive ECE, MCE, Brier Score, NLL, reliability diagram, selective-risk curve.
\end{itemize}

\vspace{0.2cm}
\textbf{Kết luận review:} Temperature Scaling là baseline mạnh, đơn giản, ít ảnh hưởng accuracy; các phương pháp adaptive/ensemble hữu ích khi gặp distribution shift.
\end{frame}

\begin{frame}{Literature Review 2: LLM Calibration}
\textbf{Vấn đề chính:} LLM có thể trả lời rất tự tin nhưng sai; RLHF và black-box API làm token probability không còn là tín hiệu calibration đáng tin tuyệt đối.

\vspace{0.2cm}
\textbf{Nhóm hướng tiếp cận đã review:}
\begin{itemize}
\item \textbf{Token/log-prob based:} dùng xác suất chuỗi sinh ra khi có logit hoặc log-prob access.
\item \textbf{Verbalized confidence:} yêu cầu mô hình tự ước lượng độ tin cậy bằng prompt.
\item \textbf{Self-consistency:} lấy nhiều mẫu trả lời, dùng mức đồng thuận làm confidence signal.
\item \textbf{RAG grounding:} kiểm tra độ bám nguồn và chất lượng retrieval để giảm hallucination.
\item \textbf{APRICOT-style:} embedding câu hỏi, clustering, tính accuracy theo cụm, huấn luyện auxiliary calibrator.
\end{itemize}
\end{frame}

\begin{frame}{Literature Review 3: VLM Calibration}
\textbf{Vấn đề chính:} VLM có hai nguồn bất định: tín hiệu thị giác và tín hiệu ngôn ngữ; câu trả lời mạch lạc có thể che giấu uncertainty từ ảnh.

\vspace{0.2cm}
\textbf{Nhóm hướng tiếp cận đã review:}
\begin{itemize}
\item \textbf{Distance-Aware Calibration:} dùng khoảng cách embedding text/visual để điều chỉnh confidence.
\item \textbf{Cross-modal decomposition:} tách confidence thành thành phần vision và language.
\item \textbf{Attention/grounding check:} dùng cross-attention hoặc grounding score để phát hiện câu trả lời không bám ảnh.
\item \textbf{Perturbation-based uncertainty:} đo variance khi thay đổi ảnh, prompt, hoặc sampling strategy.
\end{itemize}

\vspace{0.2cm}
\textbf{Kết luận review:} VLM cần calibration theo từng modality, không nên chỉ dùng softmax hoặc text confidence cuối cùng.
\end{frame}

\section{Preparation}

\begin{frame}{Metrics, Baselines, Benchmarks}
\textbf{Metrics library cần xây dựng:}
\begin{itemize}
\item ECE equal-width, Adaptive ECE/equal-mass, MCE, Brier Score, NLL.
\item Reliability diagram trước/sau calibration và selective-risk curve.
\end{itemize}

\vspace{0.2cm}
\textbf{Baseline thí nghiệm đã xác định:}
\begin{itemize}
\item Narrow AI: raw softmax, Temperature Scaling, Isotonic Regression, Beta/Dirichlet-style calibration.
\item LLM: token probability, verbalized confidence, self-consistency, APRICOT-style calibrator.
\item VLM: uncalibrated CLIP/LLaVA, DAC, cross-modal confidence decomposition.
\end{itemize}

\vspace{0.2cm}
\textbf{Benchmarks:} CIFAR-100, ImageNet-C, MMLU, TruthfulQA, HaluEval, POPE, VQA-v2, SEED-Bench.
\end{frame}

\begin{frame}{Phase 1 Implementation Assets}
\textbf{Narrow AI pipeline:}
\begin{itemize}
\item Quy trình extract logits từ CNN/ViT trước softmax.
\item Tối ưu temperature trên validation set bằng NLL.
\item So sánh calibration trước/sau bằng cùng bộ metrics.
\end{itemize}

\vspace{0.2cm}
\textbf{LLM/VLM foundation:}
\begin{itemize}
\item Thiết kế embedding pipeline cho câu hỏi bằng Sentence-BERT.
\item Chuẩn bị clustering semantic groups bằng HDBSCAN cho APRICOT-style calibration.
\item Xác định cách thu correctness labels, self-consistency score, grounding score.
\item Chuẩn bị hướng tích hợp cross-attention/visual uncertainty cho VLM.
\end{itemize}
\end{frame}

\section{Direction}

\begin{frame}{Định Hướng: Unified Calibration Framework}
\textbf{Ý tưởng chính:} xây dựng UCF như một framework thống nhất nhưng có module riêng cho từng loại mô hình.

\vspace{0.2cm}
\begin{itemize}
\item \textbf{Metrics layer:} chuẩn hóa cách đo calibration trên mọi architecture.
\item \textbf{Calibration layer:} triển khai post-hoc và training-time methods theo interface chung.
\item \textbf{Model-specific layer:} Narrow AI dùng logits; LLM dùng generation confidence; VLM dùng cross-modal signals.
\item \textbf{Monitoring layer:} theo dõi distribution shift, drift, và kích hoạt recalibration khi cần.
\item \textbf{Deployment layer:} hỗ trợ reject option, conformal prediction, human-in-the-loop cho case rủi ro cao.
\end{itemize}
\end{frame}
